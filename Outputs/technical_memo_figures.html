total_observations <- birds_joined %>% 
  select(observation_count) 
  
county_obs <- total_observations %>%
  st_join(counties %>% select(namelsad, area)) %>%  # join county info onto points
  st_drop_geometry() %>%
  group_by(namelsad) %>%
  summarize(
    total_observations = sum(observation_count, na.rm = TRUE),
    area = first(area)
  ) %>%
  mutate(
    total_observations = replace_na(total_observations, 0),
    density = total_observations / area
  ) %>%
  arrange(desc(density))

county_obs %>%
  rename(
    "County Name" = namelsad,
    "Total Observations" = total_observations,
    "Area (km²)" = area,
    "Density (#/km²)" = density
  ) %>%
  kable() %>%
  kable_styling(bootstrap_options = c("striped", "hover")) %>%
  row_spec(0, background = point_light_blue) %>% 
  save_kable(here::here('Outputs', 'figures', "county_density_table.png"), zoom = 4)
